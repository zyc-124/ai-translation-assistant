    1，创建表

                CREATE TABLE        创建关键字
                表名                表名，列名与类型
            show tables;
    2，删除表
        语法: DROP TABLE 表名；
                DROP                删除关键字
                表名
二，修改表
    1，修改表名
        语法: ALTER TABLE 旧表名 REAME 新表名 ;
                ALTER TABLE         修改表关键字
                REAME               修改表名关键字
    2，修改列名
        语法: ALTER TABLE 表名 CHEANGE COLUMN 旧列名 新列明 类型；
                ALTER TABLE         修改表关键字
                CHEANGE COLUMN      修改列名关键字
    3，修改列类型
        语法: ALTER TABLE 表名 MODIFY 列名 新类型；
                MODIFY              修改列类型关键字
    4，添加新的列
        语法: ALTER TABLE 表名 ADD COLUMN 新列名 类型；
                ADD COLUMN          添加列关键字
    5，删除指定的列
        语法: ALTER TABLE 表名 DROP COLUMN 列名；
                DROP COLUMN         删除列关键字
